cmake_minimum_required(VERSION 3.24)
project(SequoiaLiquidGlass VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus Test)
include(KDEInstallDirs6)
find_package(KWin 6.8 REQUIRED)
if(NOT KWin_VERSION VERSION_EQUAL "6.8.2")
    message(FATAL_ERROR "This experimental fork targets KWin 6.8.2; review the upstream blur API before updating.")
endif()
find_package(KF6Config REQUIRED)
find_package(KDecoration3 REQUIRED)
set(glass_sources blur.cpp liquidglass.qrc main.cpp)
kconfig_add_kcfg_files(glass_sources blurconfig.kcfgc)
kcoreaddons_add_plugin(sequoia_liquid_glass SOURCES ${glass_sources} INSTALL_NAMESPACE "kwin/effects/plugins")
set_target_properties(sequoia_liquid_glass PROPERTIES PREFIX "")
target_link_libraries(sequoia_liquid_glass PRIVATE KWin::kwin KF6::ConfigGui KDecoration3::KDecoration)

target_include_directories(sequoia_liquid_glass PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
find_package(KF6WindowSystem REQUIRED)
add_executable(liquid-glass-preview tests/preview.cpp)
target_link_libraries(liquid-glass-preview PRIVATE Qt6::Quick Qt6::DBus Qt6::Test KF6::WindowSystem)
