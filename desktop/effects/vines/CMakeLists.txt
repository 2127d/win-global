cmake_minimum_required(VERSION 3.24)
project(SequoiaVines VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml Widgets DBus)
find_package(KWin 6.8 REQUIRED)
find_package(KDecoration3 REQUIRED)
add_library(sequoiavines2 MODULE plugin.cpp jungle.cpp)
target_link_libraries(sequoiavines2 PRIVATE Qt6::Quick Qt6::Qml KWin::kwin)
set_target_properties(sequoiavines2 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/contents/ui/native2")
set_target_properties(sequoiavines2 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
file(COPY package/ DESTINATION "${CMAKE_BINARY_DIR}/package")
add_executable(render-vines render.cpp jungle.cpp)
target_link_libraries(render-vines PRIVATE Qt6::Gui)
