cmake_minimum_required(VERSION 3.24)
project(SequoiaVines VERSION 9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml Widgets DBus)
find_package(KF6Config REQUIRED)
find_package(KWin 6.8 REQUIRED)
find_package(KDecoration3 REQUIRED)
add_library(sequoiavines9 MODULE plugin.cpp jungle.cpp motion.cpp metrics.cpp)
target_link_libraries(sequoiavines9 PRIVATE Qt6::Quick Qt6::Qml Qt6::DBus KF6::ConfigCore KWin::kwin)
set_target_properties(sequoiavines9 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/contents/ui/native9")
set_target_properties(sequoiavines9 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
file(COPY package/ DESTINATION "${CMAKE_BINARY_DIR}/package")
add_executable(render-vines render.cpp jungle.cpp motion.cpp)
target_link_libraries(render-vines PRIVATE Qt6::Gui)

add_executable(test-vines-motion tests/motion.cpp motion.cpp)
target_link_libraries(test-vines-motion PRIVATE Qt6::Gui)
enable_testing()
add_test(NAME vines-motion COMMAND test-vines-motion)

find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_executable(test-vines-config tests/config.cpp)
target_link_libraries(test-vines-config PRIVATE Qt6::Widgets KF6::KCMUtils KF6::CoreAddons)
