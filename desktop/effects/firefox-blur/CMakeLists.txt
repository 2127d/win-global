cmake_minimum_required(VERSION 3.25)
project(sequoia_firefox_blur VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KWin 6.8 REQUIRED)
if(NOT KWin_VERSION VERSION_EQUAL "6.8.2" AND NOT KWin_VERSION VERSION_EQUAL "6.8.3")
    message(FATAL_ERROR "Firefox blur supports KWin 6.8.2 and 6.8.3; review the upstream blur API before updating.")
endif()
add_library(sequoia_firefox_blur MODULE main.cpp sequoia_firefox_blur.cpp sequoia_firefox_blur.h sequoia_firefox_blur_v2.qrc)
set_target_properties(sequoia_firefox_blur PROPERTIES PREFIX "" OUTPUT_NAME "sequoia_firefox_blur_v2")
set_property(TARGET sequoia_firefox_blur APPEND PROPERTY AUTOMOC_MACRO_NAMES KWIN_EFFECT_FACTORY_SUPPORTED)
target_compile_options(sequoia_firefox_blur PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_compile_definitions(sequoia_firefox_blur PRIVATE VULKAN_HPP_NO_EXCEPTIONS=1 VULKAN_HPP_ASSERT_ON_RESULT=void VULKAN_HPP_NO_SPACESHIP_OPERATOR=1 VULKAN_HPP_TYPESAFE_CONVERSION=1 VULKAN_HPP_HANDLES_MOVE_EXCHANGE=1)
target_link_libraries(sequoia_firefox_blur PRIVATE KWin::kwin Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::DBus KF6::CoreAddons KF6::ConfigCore KF6::WindowSystem)
target_link_options(sequoia_firefox_blur PRIVATE -Wl,--no-undefined)

# Keep startup guard tied to the headers used for this binary.
configure_file(enable-blur.sh.in enable-blur.sh @ONLY)

option(SEQUOIA_BUILD_TESTS "Build private-compositor rendering probe" OFF)
if(SEQUOIA_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(firefox-blur-probe tests/probe.cpp)
    target_link_libraries(firefox-blur-probe PRIVATE Qt6::Quick Qt6::DBus Qt6::Test KF6::WindowSystem)
endif()
