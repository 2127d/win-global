cmake_minimum_required(VERSION 3.24)
project(SequoiaModalDimming VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
include(KDEInstallDirs6)
find_package(KWin 6.7 REQUIRED)
find_package(KDecoration3 REQUIRED)
# Development SDKs can be extracted privately without installing system packages.
if(SEQUOIA_KWIN_INCLUDE_DIR)
    set_property(TARGET KWin::kwin PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${SEQUOIA_KWIN_INCLUDE_DIR}")
endif()
kcoreaddons_add_plugin(sequoia_modal_dimming SOURCES modaldimming.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
set_target_properties(sequoia_modal_dimming PROPERTIES PREFIX "")
target_link_libraries(sequoia_modal_dimming PRIVATE KWin::kwin Qt6::DBus KDecoration3::KDecoration)
option(SEQUOIA_BUILD_TESTS "Build private-compositor integration probe" OFF)
if(SEQUOIA_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test Qml GuiPrivate)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(ProbeWayland REQUIRED IMPORTED_TARGET wayland-client)
    add_subdirectory(client)
    add_library(modal-seam-placement MODULE tests/placement.cpp)
    target_link_libraries(modal-seam-placement PRIVATE KWin::kwin)
    add_executable(modal-dimming-probe tests/probe.cpp)
    target_link_libraries(modal-dimming-probe PRIVATE sequoia-modal-bridge Qt6::GuiPrivate PkgConfig::ProbeWayland Qt6::Quick Qt6::Qml Qt6::Test Qt6::DBus)
endif()
