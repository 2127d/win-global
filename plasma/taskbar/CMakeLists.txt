cmake_minimum_required(VERSION 3.22)
project(SequoiaTaskbar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_package(ECM 6.10 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Qml Quick Test)
find_package(KF6 REQUIRED COMPONENTS Config CoreAddons KCMUtils I18n)
add_library(taskbarsettings STATIC settings.cpp)
target_link_libraries(taskbarsettings PUBLIC Qt6::Qml KF6::ConfigCore)
target_include_directories(taskbarsettings PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(taskbarplugin MODULE plugin.cpp)
target_link_libraries(taskbarplugin PRIVATE taskbarsettings)
set_target_properties(taskbarplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/org/sequoia/taskbar")
configure_file(qmldir "${CMAKE_BINARY_DIR}/qml/org/sequoia/taskbar/qmldir" COPYONLY)
install(TARGETS taskbarplugin DESTINATION ${KDE_INSTALL_QMLDIR}/org/sequoia/taskbar)
install(FILES qmldir DESTINATION ${KDE_INSTALL_QMLDIR}/org/sequoia/taskbar)
# The taskbar and the existing Sequoia widget use the same motion implementation.
set(motion_files ../control-center/taskmanager/DockMotion.qml MotionIcon.qml)
foreach(motion_file IN LISTS motion_files)
    get_filename_component(motion_name ${motion_file} NAME)
    configure_file(${motion_file} "${CMAKE_BINARY_DIR}/qml/org/sequoia/taskbar/${motion_name}" COPYONLY)
endforeach()
install(FILES ${motion_files} DESTINATION ${KDE_INSTALL_QMLDIR}/org/sequoia/taskbar)
kcoreaddons_add_plugin(org.sequoia.taskbar SOURCES native-factory.cpp INSTALL_NAMESPACE plasma/applets)
set_target_properties(org.sequoia.taskbar PROPERTIES PREFIX "")
target_link_libraries(org.sequoia.taskbar PRIVATE KF6::CoreAddons)
add_subdirectory(kcm)
install(DIRECTORY panel/ DESTINATION ${KDE_INSTALL_DATADIR}/plasma/plasmoids/org.sequoia.taskbar)
install(DIRECTORY LICENSES/ DESTINATION ${KDE_INSTALL_DATADIR}/plasma/plasmoids/org.sequoia.taskbar/LICENSES)
install(DIRECTORY template/ DESTINATION ${KDE_INSTALL_DATADIR}/plasma/layout-templates/org.sequoia.taskbar.panel)
include(CTest)
if(BUILD_TESTING)
    add_executable(taskbar-settings-test tests/settings_test.cpp)
    target_compile_definitions(taskbar-settings-test PRIVATE TASKBAR_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(taskbar-settings-test PRIVATE taskbarsettings Qt6::Test)
    add_test(NAME taskbar-settings COMMAND taskbar-settings-test)
    add_executable(taskbar-kcm-test tests/kcm_test.cpp)
    target_link_libraries(taskbar-kcm-test PRIVATE Qt6::Quick KF6::KCMUtilsQuick KF6::CoreAddons)
    target_compile_definitions(taskbar-kcm-test PRIVATE TASKBAR_KCM_FILE="$<TARGET_FILE:kcm_sequoia_taskbar>")
    add_dependencies(taskbar-kcm-test kcm_sequoia_taskbar)
endif()
